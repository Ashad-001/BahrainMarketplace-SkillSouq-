          <div className="bg-purple-600 h-full transition-all duration-700" style={{ width: `${(subStep/4)*100}%` }} />
        </div>
      </div>
      <div className="min-h-[400px]">
        {subStep === 1 && (
          <div className="space-y-6 p-8 bg-purple-50/30 dark:bg-purple-900/10 rounded-[2.5rem] border border-purple-100 dark:border-purple-900/30">
             <div className="flex items-center gap-3 mb-2"><GraduationCap className="w-6 h-6 text-purple-600" /><h3 className="font-black text-purple-900 dark:text-purple-100 uppercase tracking-tight">Advisory Background</h3></div>
             <Input placeholder="Consulting Firm / Agency Name" value={props.firmName} onChange={(e) => props.setFirmName(e.target.value)} className="h-14 font-bold rounded-xl" />
             <Input placeholder="Lead Consultant Name" value={props.leadConsultant} onChange={(e) => props.setLeadConsultant(e.target.value)} className="h-14 font-bold rounded-xl" />
             <Input type="number" placeholder="Years of Specialized Experience" value={props.experience} onChange={(e) => props.setExperience(e.target.value)} className="h-14 font-bold rounded-xl" />
          </div>
        )}
        {subStep === 2 && (
          <div className="space-y-8">
            <div className="flex items-center gap-2 mb-2"><PieChart className="w-6 h-6 text-purple-600" /><h3 className="text-lg font-black tracking-tight">Select Core Industry Verticals</h3></div>
            <div className="flex flex-wrap gap-3">
              {industries.map(ind => (
                <button 
                  key={ind} 
                  onClick={() => props.setIndustryExpertise((prev: string[]) => prev.includes(ind) ? prev.filter(x => x !== ind) : [...prev, ind])} 
                  className={`px-8 py-4 rounded-2xl text-xs font-black transition-all border-2 flex items-center gap-2 ${props.industryExpertise.includes(ind) ? 'bg-purple-600 border-purple-600 text-white shadow-xl' : 'bg-white dark:bg-zinc-950 border-zinc-100 dark:border-zinc-800 text-zinc-500'}`}
                >
                  {props.industryExpertise.includes(ind) && <CheckCircle2 className="w-4 h-4" />}
                  {ind}
                </button>
              ))}
            </div>
          </div>
        )}
        {subStep === 3 && (
          <div className="space-y-8">
            <div>
               <div className="flex justify-between items-end mb-2">
                 <label className="text-xs font-black uppercase text-zinc-400 tracking-widest ml-1">Advisory Headline</label>
                 <button onClick={props.handleEnhanceTitle} className="text-[10px] font-black text-purple-600">✨ AI Polish</button>
               </div>
               <Input placeholder="e.g. Market Entry Strategy & Operational Optimization for Bahrain" value={props.title} onChange={(e) => props.setTitle(e.target.value)} className="h-14 font-bold rounded-2xl bg-zinc-50 dark:bg-zinc-950" />
            </div>
            <Input type="number" placeholder="Starting Engagement Fee (BHD)" value={props.price} onChange={(e) => props.setPrice(e.target.value)} className="h-14 font-bold rounded-2xl bg-zinc-50 dark:bg-zinc-950 w-full md:w-1/2" />
            <div>
               <div className="flex justify-between items-end mb-2">
                  <label className="text-xs font-black text-zinc-400 uppercase tracking-widest ml-1">Professional Bio & Methodology</label>
                  <button onClick={props.handleGenerateSummary} className="text-[10px] font-black text-purple-600 bg-purple-50 dark:bg-purple-900/20 px-3 py-1 rounded-full">✨ AI Write Bio</button>
               </div>
               <textarea placeholder="Explain your methodology, past successes, and the tangible ROI clients can expect..." value={props.aboutMe} onChange={(e) => props.setAboutMe(e.target.value)} className="w-full h-44 p-5 rounded-[2rem] bg-zinc-50 dark:bg-zinc-950 border border-zinc-200 dark:border-zinc-800 font-medium text-sm outline-none resize-none" />
            </div>
          </div>
        )}
        {subStep === 4 && <LocationSection {...props} />}
      </div>
      <div className="flex justify-between mt-12">
        <Button variant="ghost" onClick={handleBack} className="h-14 px-8 rounded-2xl font-bold">Back</Button>
        <Button onClick={handleNext} disabled={subStep === 1 && (!props.firmName || !props.leadConsultant)} className="h-14 px-12 rounded-2xl bg-purple-600 text-white font-black shadow-2xl hover:scale-105 active:scale-95 transition-all">
          {subStep === 4 ? "Launch Advisory Suite" : "Continue"} <ChevronRight className="w-5 h-5 ml-2" />
        </Button>
      </div>
    </div>
  );
};

export default function PostServicePage() {
  const [isCheckingSession, setIsCheckingSession] = useState(true);
  const [step, setStep] = useState(1);
  const [loading, setLoading] = useState(false);
  const [errorMsg, setErrorMsg] = useState("");
  const [supabase, setSupabase] = useState<any>(null);
  const [user, setUser] = useState<any>(null);

  // AI State
  const [isGeneratingTitle, setIsGeneratingTitle] = useState(false);
  const [isGeneratingSummary, setIsGeneratingSummary] = useState(false);

  // SHARED STATE
  const [category, setCategory] = useState("");
  const [title, setTitle] = useState("");
  const [price, setPrice] = useState("");
  const [experience, setExperience] = useState("");
  const [location, setLocation] = useState("");
  const [aboutMe, setAboutMe] = useState("");
  const [firmName, setFirmName] = useState("");

  // LEGAL SPECIFIC
  const [accreditation, setAccreditation] = useState("");
  const [idFile, setIdFile] = useState<File | null>(null);
  const [licenseFile, setLicenseFile] = useState<File | null>(null);

  // IT SPECIFIC
  const [idNumber, setIdNumber] = useState("");
  const [contactEmail, setContactEmail] = useState("");
  const [portfolioUrl, setPortfolioUrl] = useState("");
  const [techStack, setTechStack] = useState("");
  const [gigMediaFile, setGigMediaFile] = useState<File | null>(null);

  // CLEARANCE SPECIFIC
  const [crNumber, setCrNumber] = useState("");
  const [ministryExpertise, setMinistryExpertise] = useState<string[]>([]);

  // CONSULTING SPECIFIC
  const [leadConsultant, setLeadConsultant] = useState("");
  const [industryExpertise, setIndustryExpertise] = useState<string[]>([]);

  const categories = [
    { id: "Legal", icon: Scale, desc: "Lawyers & Legal Counsel" },
    { id: "Consulting", icon: BarChart3, desc: "Strategy & Advisory" },
    { id: "Clearance", icon: FileText, desc: "Gov & Customs Clearance" },
    { id: "IT & Tech", icon: Zap, desc: "Software & Digital Skills" },
  ];

  // GEMINI AI CALL
  const callGemini = async (prompt: string) => {
    const apiKey = ""; // Context-provided
    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
    const payload = {
      contents: [{ parts: [{ text: prompt }] }],
      systemInstruction: { parts: [{ text: "You are an elite Bahrain-based professional copywriter for Skill Souq. Respond with direct, persuasive, high-authority text. Use Kingdom of Bahrain context where relevant. No quotes or markdown headers." }] }
    };

    try {
      const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) });
      const data = await response.json();
      return data.candidates?.[0]?.content?.parts?.[0]?.text?.trim() || "";
    } catch (e) { console.error(e); return ""; }
  };

  const handleEnhanceTitle = async () => {
    if (!title) return;
    setIsGeneratingTitle(true);
    const result = await callGemini(`Improve this ${category} service headline to sound elite, trustworthy and premium for the Bahraini market: "${title}". Return ONLY the resulting headline.`);
    if (result) setTitle(result);
    setIsGeneratingTitle(false);
  };

  const handleGenerateSummary = async () => {
    if (!title) return;
    setIsGeneratingSummary(true);
    const prompt = `Write a 2-paragraph professional profile bio for a ${category} specialist on Skill Souq. 
    Name/Firm: ${firmName}. 
    Headline: "${title}". 
    Experience: ${experience} years. 
    Location: ${location || 'Bahrain'}. 
    ${category === 'IT & Tech' ? 'Tech Stack: ' + techStack : ''}
    Focus on expertise, ROI for the client, and authoritative presence.`;
    const result = await callGemini(prompt);
    if (result) setAboutMe(result);
    setIsGeneratingSummary(false);
  };

  useEffect(() => {
    const init = async () => {
      if (!(window as any).supabase) {
        const script = document.createElement("script");
        script.src = "https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2";
        script.async = true;
        script.onload = () => createInstance();
        document.head.appendChild(script);
      } else { createInstance(); }
    };

    const createInstance = async () => {
      const PROJECT_URL = "https://jgrovqdvsightthcdxaj.supabase.co";
      const PROJECT_ANON_KEY = "sb_publishable_iuxeDzeSzyUDR3RIZiqe2w_93FWUS8J";
      if ((window as any).supabase) {
        const client = (window as any).supabase.createClient(PROJECT_URL, PROJECT_ANON_KEY);
        setSupabase(client);
        const { data: { session } } = await client.auth.getSession();
        if (session) {
          setUser(session.user);
          const userInterests = session.user.user_metadata?.interests || [];
          const matched = categories.find(c => userInterests.includes(c.id));
          if (matched) { setCategory(matched.id); setStep(2); }
        } else { window.location.href = "/login"; }
        setIsCheckingSession(false);
      }
    };
    init();
  }, []);

  const handlePublish = async () => {
    if (!supabase || !user) return;
    setLoading(true);
    setErrorMsg("");

    const finalName = firmName || `${user?.user_metadata?.first_name || 'Professional'}`;
    let compiledDescription = `### Summary\n${aboutMe}\n\n`;
    
    if (category === "Legal") compiledDescription += `**Accreditation:** ${accreditation}\n**Location:** ${location}\n**Verification:** Documents Securely Submitted.`;
    else if (category === "IT & Tech") compiledDescription += `**Stack:** ${techStack}\n**Portfolio:** ${portfolioUrl}\n**Contact:** ${contactEmail}\n**Experience:** ${experience} Years`;
    else if (category === "Clearance") compiledDescription += `**Ministries:** ${ministryExpertise.join(", ")}\n**CR:** ${crNumber}\n**Location:** ${location}`;
    else if (category === "Consulting") compiledDescription += `**Lead:** ${leadConsultant}\n**Industries:** ${industryExpertise.join(", ")}\n**Location:** ${location}`;

    try {
      const { error } = await supabase.from('gigs').insert([{
        seller_id: user.id,
        title: title || `${category} Specialized Service`, 
        category,
        price: parseFloat(price) || 0,
        description: compiledDescription
      }]);
      if (error) throw error;
      setStep(4); 
    } catch (err: any) { setErrorMsg(err.message || "Failed to publish."); } 
    finally { setLoading(false); }
  };

  const formProps = {
    title, setTitle, firmName, setFirmName, accreditation, setAccreditation,
    idFile, setIdFile, licenseFile, setLicenseFile, experience, setExperience,
    idNumber, setIdNumber, contactEmail, setContactEmail, portfolioUrl, setPortfolioUrl, 
    techStack, setTechStack, gigMediaFile, setGigMediaFile,
    crNumber, setCrNumber, ministryExpertise, setMinistryExpertise,
    leadConsultant, setLeadConsultant, industryExpertise, setIndustryExpertise,
    price, setPrice, location, setLocation, aboutMe, setAboutMe,
    handleEnhanceTitle, handleGenerateSummary, isGeneratingTitle, isGeneratingSummary,
    onBack: () => setStep(1), onNext: handlePublish
  };

  if (isCheckingSession) return (
    <div className="min-h-screen flex items-center justify-center bg-zinc-50 dark:bg-zinc-950 font-sans">
       <div className="flex flex-col items-center">
         <Loader2 className="w-10 h-10 animate-spin text-blue-600 mb-4" />
         <p className="text-xs font-black text-zinc-400 uppercase tracking-widest animate-pulse">Synchronizing Workspace...</p>
       </div>
    </div>
  );

  return (
    <div className="min-h-screen bg-zinc-50 dark:bg-zinc-950 py-10 px-6 pb-32">
      <div className="max-w-4xl mx-auto">
        
        {step < 4 && (
          <div className="flex items-center justify-between mb-10">
            <a href="/seller-dashboard" className="flex items-center gap-2 text-sm font-bold text-zinc-500 hover:text-zinc-900 transition-colors">
              <ArrowLeft className="w-4 h-4" /> Back to Dashboard
            </a>
            <div className="text-xs font-black text-blue-600 bg-blue-50 dark:bg-blue-900/30 px-4 py-2 rounded-full uppercase tracking-widest border border-blue-100 dark:border-blue-900/50">
              Stage {step} of 3
            </div>
          </div>
        )}

        <div className="bg-white dark:bg-zinc-900 rounded-[3rem] p-10 md:p-16 shadow-[0_32px_80px_-16px_rgba(0,0,0,0.08)] border border-zinc-200 dark:border-zinc-800 relative overflow-hidden">
          {errorMsg && <div className="mb-8 p-5 bg-red-50 dark:bg-red-900/10 text-red-600 rounded-2xl font-bold flex items-center gap-3 border border-red-100 dark:border-red-900/30 animate-in shake duration-500"><AlertCircle className="w-5 h-5"/> {errorMsg}</div>}

          {step === 1 && (
            <div className="animate-in fade-in slide-in-from-right-8 duration-500 text-center md:text-left">
              <h1 className="text-4xl font-black mb-4 tracking-tighter">Choose Specialization</h1>
              <p className="text-zinc-500 font-medium mb-12">This unlocks the specific legal or technical requirements needed for your profile.</p>
              
              <div className="grid grid-cols-1 md:grid-cols-2 gap-5">
                {categories.map((cat) => (
                  <button key={cat.id} onClick={() => setCategory(cat.id)} className={`p-8 rounded-[2.5rem] border-2 text-left transition-all group ${category === cat.id ? 'border-blue-600 bg-blue-50/50 dark:bg-blue-900/10' : 'border-zinc-100 dark:border-zinc-800 hover:border-blue-200'}`}>
                    <div className={`w-14 h-14 rounded-2xl flex items-center justify-center mb-6 transition-transform group-hover:scale-110 ${category === cat.id ? 'bg-blue-600 text-white shadow-xl' : 'bg-zinc-100 dark:bg-zinc-800 text-zinc-400'}`}>
                       <cat.icon className="w-7 h-7" />
                    </div>
                    <h3 className="text-xl font-black mb-1">{cat.id}</h3>
                    <p className="text-xs font-bold text-zinc-400 leading-relaxed uppercase tracking-tight">{cat.desc}</p>
                  </button>
                ))}
              </div>
              <div className="mt-16 flex justify-end">
                <Button onClick={() => setStep(2)} disabled={!category} className="h-16 px-12 rounded-[2rem] bg-zinc-900 dark:bg-white text-white dark:text-zinc-900 font-black shadow-2xl hover:scale-105 active:scale-95 transition-all">
                   Continue: Step 2 <ChevronRight className="ml-2 w-5 h-5"/>
                </Button>
              </div>
            </div>
          )}

          {step === 2 && category === "Legal" && <LegalForm {...formProps} />}
          {step === 2 && category === "IT & Tech" && <ITTechForm {...formProps} />}
          {step === 2 && category === "Clearance" && <ClearanceForm {...formProps} />}
          {step === 2 && category === "Consulting" && <ConsultingForm {...formProps} />}

          {step === 4 && (
            <div className="text-center py-10 animate-in zoom-in duration-700">
              <div className="w-36 h-36 mx-auto bg-emerald-100 dark:bg-emerald-900/20 rounded-[3rem] flex items-center justify-center mb-10 rotate-6 animate-pulse">
                <CheckCircle2 className="w-16 h-16 text-emerald-600" />
              </div>
              <h1 className="text-5xl font-black tracking-tighter mb-6 text-zinc-900 dark:text-white leading-none">Service Active!</h1>
              <p className="text-zinc-500 text-xl font-medium mb-12 max-w-sm mx-auto">Your specialized profile has been saved and is now live in the marketplace.</p>
              <div className="flex flex-col sm:flex-row justify-center gap-4">
                 <a href="/seller-dashboard" className="h-16 px-10 rounded-[2rem] border-2 border-zinc-200 dark:border-zinc-800 flex items-center justify-center font-black transition-all hover:bg-zinc-50">Back to Dashboard</a>
                 <a href="/marketplace" className="h-16 px-10 rounded-[2rem] bg-zinc-900 dark:bg-white text-white dark:text-zinc-900 flex items-center justify-center font-black hover:scale-105 active:scale-95 transition-all shadow-xl">View Marketplace</a>
              </div>
            </div>
          )}

        </div>
      </div>
    </div>
  );
